const http = require('http'); const fs = require('fs'); const path = require('path'); const PORT = Number(process.env.PORT || 10000); const ROOT = __dirname; const ADMIN_PASSWORD = process.env.ADMIN_PASSWORD || ''; let latestMessage = { text: '', name: '', updatedAt: null, id: 0 }; let nextId = 1; let visitorMessages = []; let nextVisitorId = 1; const mime = { '.html': 'text/html; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.css': 'text/css; charset=utf-8' }; function send(res, status, body, type = 'text/plain; charset=utf-8') { res.writeHead(status, { 'Content-Type': type, 'Cache-Control': 'no-store' }); res.end(body); } function json(res, status, data) { send(res, status, JSON.stringify(data), 'application/json; charset=utf-8'); } function readJson(req, res, callback) { let body = ''; req.on('data', chunk => { body += chunk; if (body.length > 10000) { send(res, 413, 'Mensagem muito grande.'); req.destroy(); } }); req.on('end', () => { if (res.writableEnded) return; try { callback(JSON.parse(body || '{}')); } catch { json(res, 400, { error: 'Pedido inválido.' }); } }); req.on('error', () => { if (!res.writableEnded) { json(res, 400, { error: 'Erro ao receber pedido.' }); } }); } http.createServer((req, res) => { const url = new URL( req.url, `http://${req.headers.host || 'localhost'}` ); // LOGIN DO ADMINISTRADOR if (req.method === 'POST' && url.pathname === '/api/login') { return readJson(req, res, data => { if (!ADMIN_PASSWORD) { return json(res, 503, { error: 'Configure ADMIN_PASSWORD no Render.' }); } if (data.password !== ADMIN_PASSWORD) { return json(res, 401, { error: 'Senha incorreta.' }); } return json(res, 200, { ok: true }); }); } // VISITANTES LEEM O AVISO DO ADMINISTRADOR if (req.method === 'GET' && url.pathname === '/api/message') { return json(res, 200, latestMessage); } // ADMINISTRADOR PUBLICA AVISO if (req.method === 'POST' && url.pathname === '/api/message') { if (!ADMIN_PASSWORD) { return json(res, 503, { error: 'O administrador ainda não configurou uma senha.' }); } if (req.headers['x-admin-password'] !== ADMIN_PASSWORD) { return json(res, 401, { error: 'Acesso negado.' }); } return readJson(req, res, data => { const text = String(data.text || '').trim().slice(0, 500); const name = String(data.name || 'Administrador') .trim() .slice(0, 60); if (!text) { return json(res, 400, { error: 'Escreva uma mensagem.' }); } latestMessage = { text, name: name || 'Administrador', updatedAt: new Date().toISOString(), id: nextId++ }; return json(res, 200, { ok: true }); }); } // VISITANTE ENVIA MENSAGEM if (req.method === 'POST' && url.pathname === '/api/visitor-message') { return readJson(req, res, data => { const name = String(data.name || '').trim().slice(0, 60); const text = String(data.text || '').trim().slice(0, 500); if (!name) { return json(res, 400, { error: 'Digite seu nome.' }); } if (!text) { return json(res, 400, { error: 'Digite uma mensagem.' }); } const message = { id: nextVisitorId++, name, text, createdAt: new Date().toISOString() }; visitorMessages.push(message); // Mantém somente as últimas 100 mensagens if (visitorMessages.length > 100) { visitorMessages = visitorMessages.slice(-100); } return json(res, 201, { ok: true, message }); }); } // ADMINISTRADOR CONSULTA MENSAGENS DOS VISITANTES if (req.method === 'GET' && url.pathname === '/api/visitor-messages') { if (!ADMIN_PASSWORD) { return json(res, 503, { error: 'Senha administrativa não configurada.' }); } if (req.headers['x-admin-password'] !== ADMIN_PASSWORD) { return json(res, 401, { error: 'Acesso negado.' }); } return json(res, 200, { messages: visitorMessages }); } // ENTREGA OS ARQUIVOS DO SITE const route = url.pathname === '/' ? '/index.html' : url.pathname; const file = path.resolve(ROOT, '.' + route); if (!file.startsWith(ROOT + path.sep)) { return send(res, 403, 'Proibido.'); } fs.readFile(file, (err, data) => { if (err) { return send(res, 404, 'Página não encontrada.'); } return send( res, 200, data, mime[path.extname(file)] || 'application/octet-stream' ); }); }).listen(PORT, '0.0.0.0', () => { console.log('Wi-Fi Messenger rodando na porta ' + PORT); });
